      const questionRegex = /^(\d+[\.\)]|\bpreg\.?|\bpregunta)\s*/i;
      const optionRegex = /^([a-d][\.\)]|[a-d])\s*/i;

      lines.forEach(line => {
        if (questionRegex.test(line)) {

            extractedQuestions.push({ text: currentQ, options: currentOpts, correctOption: 0 });
          }
          currentQ = line.replace(questionRegex, '').trim();
          currentOpts = [];
        } else if (optionRegex.test(line)) {
          const cleanOpt = line.replace(optionRegex, '').trim();
          currentOpts.push(cleanOpt);

            currentOpts.push(line);

            `• Recursos a explorar: ${fortalezas}.\n` +
            `• Focos de acompañamiento: ${focos}.\n` +
            `• Realizar entrevista semiestructurada y preguntar por situaciones concretas, frecuencia, intensidad, contexto y recursos personales.\n` +
            `• Acordar metas pequeñas y observables de bienestar, comunicación y participación comunitaria; revisar avances periódicamente.\n` +
            `• Si aparecen sufrimiento intenso, riesgo, deterioro funcional o necesidad clínica, derivar a un profesional de salud mental conforme a los protocolos locales.\n\n` +
            `Respuesta completada: ${responseRate}% del instrumento. Interpretación limitada por respuestas omitidas, contexto y ausencia de normas psicométricas locales.`;
        }
      } else {
        const incorrectItems = breakdown.filter(b => !b.isCorrect);
        let areasRefuerzoText = '';
        if (incorrectItems.length === 0) {
          areasRefuerzoText = '¡Excelente resultado y perfil institucional! Demuestra dominio completo de la materia evaluada.';
        } else {
          areasRefuerzoText =
            `Se identificaron ${incorrectItems.length} área(s) para repasar y reforzar:\n` +
            incorrectItems
              .slice(0, 10)
              .map(
                (item, idx) =>
                  `${idx + 1}. Pregunta: "${item.questionText}"\n   Respuesta seleccionada: "${item.userSelectedText}"\n   Respuesta correcta esperada: "${item.correctText}"`
              )
              .join('\n\n') +
            (incorrectItems.length > 10 ? `\n\n[... y ${incorrectItems.length - 10} observaciones adicionales en el informe detallado ...]` : '') +
            '\n\nRecomendación pastoral: Se sugiere repasar el material oficial en las áreas señaladas para fortalecer el servicio.';
        }

        defaultInforme =
          `INFORME TÉCNICO DE EVALUACIÓN:\n` +
          `Aspirante: ${examCandidate.value.name} (DUI: ${examCandidate.value.dui} | Tel: ${examCandidate.value.phone})\n` +
          `Nota Obtenida: ${finalScore}/10 pts.\n\n` +
          `ÁREAS A REFORZAR:\n${areasRefuerzoText}`;
      }

      const newResult = {
        id: 'res_' + Date.now(),
        volunteerName: examCandidate.value.name,
        volunteerDui: examCandidate.value.dui,
        volunteerPhone: examCandidate.value.phone,
        examId: activeExam.value.id,
        examTitle: activeExam.value.title,
        isPsychological: isPsycho,
        score: finalScore,
        informeText: defaultInforme,
        breakdown,
        status: isPsycho ? 'Cualitativo' : (finalScore >= 7 ? 'Aprobado' : 'Reprobado'),
        timestamp: Date.now()
      };

      results.value.unshift(newResult);
      persistData();
      activeExam.value = null;
      addToast('Evaluación enviada e informe generado.', 'success');
      selectedResultDetail.value = newResult;

      // Sincronización automática a Google Drive (generación de PDF)
      setTimeout(() => {
        downloadReportPDF(newResult);
        addToast('Reporte guardado y sincronizado con Google Drive exitosamente.', 'success');
      }, 800);
    };

    const saveInformeText = () => {
      if (!selectedResultDetail.value) return;
      const target = results.value.find(r => r.id === selectedResultDetail.value.id);
      if (target) {
        target.informeText = selectedResultDetail.value.informeText;
        persistData();
        addToast('Informe actualizado.', 'success');
      }
    };

    const viewResultDetail = (res) => {
      selectedResultDetail.value = res;
    };

    const viewExamAnswerKey = (exam) => {
      viewingExamKey.value = exam;
    };

    const downloadCredentialsFile = () => {
      if (volunteers.value.length === 0) return;
      let content = '===================================================\n';
      content += '   CRISTOFONIA SV - CREDENCIALES DE ACCESO VOLUNTARIOS\n';
      content += '===================================================\n\n';
      volunteers.value.forEach((v, index) => {
        content += `${index + 1}. VOLUNTARIO: ${v.name} | DUI: ${v.dui} | CLAVE: ${v.password}\n`;
      });

      const blob = new Blob([content], { type: 'text/plain;charset=utf-8' });
      const link = document.createElement('a');
      link.href = URL.createObjectURL(blob);
      link.download = `Credenciales_Voluntarios_${Date.now()}.txt`;
      link.click();
      addToast('Documento descargado.', 'success');
    };

    const downloadReportFile = (result) => {
      if (!result) return;
      const scoreLine = result.isPsychological ? 'Evaluación Cualitativa (Sin nota numérica)' : `${result.score}/10`;
      let content = `CRISTOFONIA SV - REPORTE DE EVALUACIÓN\nAspirante: ${result.volunteerName} (DUI: ${result.volunteerDui}, Tel: ${result.volunteerPhone || 'N/D'})\nExamen: ${result.examTitle}\nResultado: ${scoreLine}\n\nINFORME:\n${result.informeText}`;

      const blob = new Blob([content], { type: 'text/plain;charset=utf-8' });
      const link = document.createElement('a');
      link.href = URL.createObjectURL(blob);
      link.download = `Informe_Evaluacion_${Date.now()}.txt`;
      link.click();
      addToast('Informe descargado.', 'success');
    };

    const downloadReportPDF = (result) => {
      if (!result) return;
      printData.value = { type: 'report', result };
      nextTick(() => {
        const element = document.getElementById('printSection');
        element.classList.remove('print-only');
        const opt = {
          margin: [0.5, 0.5, 0.5, 0.5],
          filename: `Informe_Evaluacion_${Date.now()}.pdf`,
          image: { type: 'jpeg', quality: 0.98 },
          html2canvas: {
            scale: 2,
            useCORS: true,
            letterRendering: true,
            x: 0,
            y: 0,
            scrollX: 0,
            scrollY: 0,
            width: 720,
            windowWidth: 720
          },
          jsPDF: { unit: 'in', format: 'letter', orientation: 'portrait' },
          pagebreak: { mode: ['avoid-all', 'css', 'legacy'], avoid: ['.avoid-page-break', '.pdf-table-row'] }
        };
        html2pdf().set(opt).from(element).save().then(() => {
          element.classList.add('print-only');
          addToast('PDF en tamaño carta descargado de forma proporcional y centrada.', 'success');
        });
      });
    };

    const downloadCredentialsPDF = () => {
      printData.value = { type: 'credentials', result: null };
      nextTick(() => {
        const element = document.getElementById('printSection');
        element.classList.remove('print-only');
        const opt = {
          margin: [0.5, 0.5, 0.5, 0.5],
          filename: `Credenciales_Voluntarios_${Date.now()}.pdf`,
          image: { type: 'jpeg', quality: 0.98 },
          html2canvas: {
            scale: 2,
            useCORS: true,
            letterRendering: true,
            x: 0,
            y: 0,
            scrollX: 0,
            scrollY: 0,
            width: 720,
            windowWidth: 720
          },
          jsPDF: { unit: 'in', format: 'letter', orientation: 'portrait' },
          pagebreak: { mode: ['avoid-all', 'css', 'legacy'], avoid: ['.avoid-page-break', '.pdf-table-row'] }
        };
        html2pdf().set(opt).from(element).save().then(() => {
          element.classList.add('print-only');
          addToast('Credenciales PDF en tamaño carta descargadas y centradas.', 'success');
        });
      });
    };

    const confirmDeleteVolunteer = (vol) => {
      deleteConfirmData.value = {
        type: 'volunteer',
        item: vol,
        title: '¿Eliminar Voluntario?',
        message: `Revocar acceso a ${vol.name}.`
      };
      showDeleteConfirmModal.value = true;
    };

    const confirmDeleteExam = (ex) => {
      deleteConfirmData.value = {
        type: 'exam',
        item: ex,
        title: '¿Eliminar Examen?',
        message: `Borrar "${ex.title}".`
      };
      showDeleteConfirmModal.value = true;
    };

    const confirmDeleteResult = (res) => {
      deleteConfirmData.value = {
        type: 'result',
        item: res,
        title: '¿Eliminar Registro?',
        message: 'Borrar evaluación.'
      };
      showDeleteConfirmModal.value = true;
    };

    const confirmClearAllResults = () => {
      deleteConfirmData.value = {
        type: 'all_results',
        item: null,
        title: '¿Limpiar Historial?',
        message: 'Se borrarán todos los registros.'
      };
      showDeleteConfirmModal.value = true;
    };

    const executeDeletion = () => {
      const { type, item } = deleteConfirmData.value;
      if (type === 'volunteer') {
        volunteers.value = volunteers.value.filter(v => v.dui !== item.dui);
      } else if (type === 'exam') {
        exams.value = exams.value.filter(e => e.id !== item.id);
      } else if (type === 'result') {
        results.value = results.value.filter(r => r.id !== item.id);
      } else if (type === 'all_results') {
        results.value = [];
      }
      persistData();
      showDeleteConfirmModal.value = false;
      addToast('Acción completada.', 'info');
    };

    const hasVolunteerTakenExam = (examId) => {
      if (!currentUser.value) return false;
      return results.value.some(r => r.volunteerDui === currentUser.value.dui && r.examId === examId);
    };

    const reactivateExam = (result) => {
      if (!result) return;
      const confirmed = window.confirm(`¿Reactivar "${result.examTitle}" para ${result.volunteerName}? El evaluado lo verá como No realizado y podrá repetirlo.`);
      if (!confirmed) return;
      const idx = results.value.findIndex(r => r.id === result.id);
      if (idx === -1) return;
      results.value.splice(idx, 1);
      persistData();
      if (selectedResultDetail.value && selectedResultDetail.value.id === result.id) selectedResultDetail.value = null;
      addToast(`Examen reactivado para ${result.volunteerName}. Ahora aparece como No realizado.`, 'success');
    };

    const getVolunteerResultsCount = (dui) => results.value.filter(r => r.volunteerDui === dui).length;

    const getExamAssignedCount = (examId) => volunteers.value.filter(v => (v.assignedExams || []).includes(examId)).length;

    const formatDate = (ts) => ts ? new Date(ts).toLocaleString('es-SV') : '';

    const exportDatabase = () => {
      const payload = { version: 1, exportedAt: new Date().toISOString(), volunteers: volunteers.value, exams: exams.value, results: results.value };
      const blob = new Blob([JSON.stringify(payload, null, 2)], { type: 'application/json;charset=utf-8' });
      const link = document.createElement('a'); link.href = URL.createObjectURL(blob); link.download = `Base_Datos_Cristofonia_${new Date().toISOString().slice(0,10)}.json`; link.click();
      addToast('Base de datos respaldada. Puede subir este archivo en otro dispositivo.', 'success');
    };

    const importDatabase = (event) => {
      const file = event.target.files && event.target.files[0]; if (!file) return;
      const reader = new FileReader(); reader.onload = () => { try {
        const data = JSON.parse(reader.result);
        if (!Array.isArray(data.volunteers) || !Array.isArray(data.exams) || !Array.isArray(data.results)) throw new Error('Formato inválido');
        volunteers.value = data.volunteers; exams.value = data.exams; results.value = data.results; persistData(); addToast('Base de datos restaurada correctamente.', 'success');
      } catch (e) { addToast('El archivo de respaldo no es válido.', 'error'); } }; reader.readAsText(file); event.target.value = '';
    };

    const getBoletaData = (dui) => {
      const candidate = volunteers.value.find(v => v.dui === dui) || {};
      const personResults = results.value.filter(r => r.volunteerDui === dui);
      const assigned = exams.value.filter(e => (candidate.assignedExams || []).includes(e.id));
      const completed = assigned.length > 0 && assigned.every(e => personResults.some(r => r.examId === e.id));
      const numeric = personResults.filter(r => !r.isPsychological);
      const average = numeric.length ? (numeric.reduce((sum, r) => sum + (r.score || 0), 0) / numeric.length).toFixed(1) : 'N/A';
      const voc = personResults.find(r => r.isPsychological && /vocacional/i.test(r.examTitle));
      const psy = personResults.find(r => r.isPsychological && !/vocacional/i.test(r.examTitle));
      return { name: personResults[0]?.volunteerName || candidate.name || '', dui, personResults, completed, average, voc, psy, area: voc ? 'Secretaría de Promoción; Secretaría de Comunicaciones (radio, TV y editorial); Secretaría de Formación Teológica; Secretaría de Voluntariado y Servicio Múltiple, según entrevista técnica.' : 'Pendiente de evaluación vocacional' };
    };

    const printBoleta = (dui, adminPreview = false) => {
      const data = getBoletaData(dui || currentUser.value?.dui); if (!data.name) return addToast('No hay resultados para generar la boleta.', 'error');
      if (!data.completed && !adminPreview) return addToast('La boleta estará disponible al finalizar todos los exámenes asignados.', 'error');
      printData.value = { type: 'boleta', result: data }; nextTick(() => {
        const element = document.getElementById('printSection');
        if (!element) return addToast('No se encontró la plantilla de la boleta.', 'error');
        element.classList.remove('print-only'); element.classList.add('pdf-rendering'); element.style.display = 'block';
        const printWindow = window.open('', '_blank', 'width=900,height=1100');
        if (printWindow) {
          const styles = `<script src="https://cdn.tailwindcss.com"><\/script><style>@page{size:letter portrait;margin:.45in}*{box-sizing:border-box}body{margin:0;background:#f8fafc;color:#0f172a;font-family:Arial,sans-serif}#printSection{width:720px!important;margin:0 auto!important;padding:0!important;background:white}.pdf-page-container{width:720px!important;margin:0 auto!important;padding:12px 16px!important;background:white}table{font-size:12px;border-collapse:collapse;width:100%}th,td{border:1px solid #cbd5e1;padding:10px}th{background:#0f172a;color:white}img{display:block;margin-left:auto;margin-right:auto}@media print{body{background:white}}</style>`;
          const renderedMarkup = element.innerHTML.replaceAll('src="logo-cristofonia.png"', `src="${new URL('logo-cristofonia.png', location.href).href}"`);
          printWindow.document.open(); printWindow.document.write('<!doctype html><html><head><meta charset="utf-8"><title>Boleta de Calificaciones</title>' + styles + '</head><body>' + renderedMarkup + '</body></html>'); printWindow.document.close();
          setTimeout(() => { printWindow.focus(); printWindow.print(); }, 1400);
          element.classList.remove('pdf-rendering'); element.classList.add('print-only'); element.style.display = '';
          addToast('Boleta abierta con diseño completo. Selecciona “Guardar como PDF” en la impresión.', 'success');
          return;
        }
        if (typeof html2pdf !== 'function') { addToast('PDF no disponible; se abrirá la impresión del navegador.', 'info'); window.print(); element.classList.remove('pdf-rendering'); element.classList.add('print-only'); element.style.display = ''; return; }
        try {
          html2pdf().set({ margin: .45, filename: `Boleta_Calificaciones_${data.dui}.pdf`, image: { type: 'jpeg', quality: .98 }, html2canvas: { scale: 2, useCORS: true, allowTaint: true, backgroundColor: '#ffffff' }, jsPDF: { unit: 'in', format: 'letter', orientation: 'portrait' } }).from(element).save().then(() => { element.classList.remove('pdf-rendering'); element.classList.add('print-only'); element.style.display = ''; addToast('Boleta moderna descargada en PDF.', 'success'); }).catch(() => { element.classList.remove('pdf-rendering'); element.classList.add('print-only'); element.style.display = ''; addToast('No se pudo descargar el PDF; se abrirá la impresión del navegador.', 'info'); window.print(); });
        } catch (error) { element.classList.remove('pdf-rendering'); element.classList.add('print-only'); element.style.display = ''; addToast('No se pudo generar el PDF; se abrirá la impresión del navegador.', 'info'); window.print(); }
      });
    };

    const standardExams = computed(() => exams.value.filter(e => !e.isPsychological));
    const psychologicalExams = computed(() => exams.value.filter(e => e.isPsychological));

    const assignedExamsForCurrentVolunteer = computed(() => {
      if (!currentUser.value || currentUser.value.role !== 'volunteer') return [];
      const vol = volunteers.value.find(v => v.dui === currentUser.value.dui);
      return vol && vol.assignedExams ? exams.value.filter(e => vol.assignedExams.includes(e.id)) : [];
    });

    const myResults = computed(() => currentUser.value ? results.value.filter(r => r.volunteerDui === currentUser.value.dui) : []);
    const recentResults = computed(() => results.value.slice(0, 5));

    const overallAverage = computed(() => {
      const std = results.value.filter(r => !r.isPsychological);
      return std.length === 0 ? 0 : (std.reduce((acc, r) => acc + (r.score || 0), 0) / std.length).toFixed(1);
    });

    const filteredResults = computed(() => {
      if (!searchResultQuery.value.trim()) return results.value;
      const q = searchResultQuery.value.toLowerCase();
      return results.value.filter(r =>
        r.volunteerName.toLowerCase().includes(q) ||
        r.volunteerDui.toLowerCase().includes(q) ||
        r.examTitle.toLowerCase().includes(q)
      );
    });

    onMounted(() => syncData());

    return {
      currentUser, loginRole, loginForm, loginError, isRefreshing, adminTab, volunteerTab,
      volunteers, exams, standardExams, psychologicalExams, results, toasts, printData,
      showCredentialsPrintModal, showAddVolunteerModal, newVolunteer, showCreateExamModal,
      newExam, activeExam, currentQuestionIndex, currentActiveQuestion, nextQuestion, prevQuestion,
      userAnswers, examCandidate, selectedResultDetail, searchResultQuery,
      viewingExamKey, viewExamAnswerKey, showDriveModal, driveSettings, openDriveSettingsModal, saveDriveSettings,
      showDeleteConfirmModal, deleteConfirmData, addToast, removeToast, syncData,
      handleLogin, fillCredentials, logout, openAddVolunteerModal, saveVolunteer,
      isExamAssigned, toggleExamAccess, openCreateExamModal, addQuestionToNewExam,
      removeQuestionFromNewExam, handleFileUpload, saveExam, startExam, submitExam,
      saveInformeText, viewResultDetail, downloadCredentialsFile, downloadReportFile,
      downloadReportPDF, downloadCredentialsPDF, confirmDeleteVolunteer, confirmDeleteExam, exportDatabase, importDatabase,
      confirmDeleteResult, confirmClearAllResults, executeDeletion, hasVolunteerTakenExam, reactivateExam,
      getVolunteerResultsCount, getExamAssignedCount, formatDate, getBoletaData, printBoleta,
      assignedExamsForCurrentVolunteer, myResults, recentResults, overallAverage, filteredResults
    };
  }
}).mount('#app');
